    function adjustZoom(delta) {

      var nextScale = currentScale + delta;

      if (nextScale < minScale) nextScale = minScale;

      if (nextScale > maxScale) nextScale = maxScale;

      currentScale = nextScale;

      if (currentScale === 1) { posX = 0; posY = 0; }

      applyImageTransform();

    }


    function applyImageTransform() {

      var img = document.getElementById('zoomedImg');

      if (!img) return;

    function initZoomEvents() {

      var viewport = document.getElementById('zoomViewport');

      if (!viewport || viewport.dataset.eventsBound === "true") return;

      viewport.dataset.eventsBound = "true";


      viewport.addEventListener('touchstart', function(e) {

        if (e.touches.length === 2) {

          isDragging = false;

          startDist = Math.hypot(

            e.touches[0].clientX - e.touches[1].clientX,

            e.touches[0].clientY - e.touches[1].clientY

          );

          initialScale = currentScale;

        } else if (e.touches.length === 1) {

          var currentTime = new Date().getTime();

          var tapGap = currentTime - lastTapTime;

          

          if (tapGap < 300 && tapGap > 0) {

            if (currentScale > 1) {

              resetZoomState();

            } else {

              currentScale = 2.5;

              applyImageTransform();

            }

            lastTapTime = 0;

            return;

          }

          lastTapTime = currentTime;


          if (currentScale > 1) {

            isDragging = true;

            startX = e.touches[0].clientX - posX;

            startY = e.touches[0].clientY - posY;

          }

        }

      }, { passive: false });


      viewport.addEventListener('touchmove', function(e) {

        e.preventDefault();


        if (e.touches.length === 2) {

          var dist = Math.hypot(

            e.touches[0].clientX - e.touches[1].clientX,

            e.touches[0].clientY - e.touches[1].clientY

          );

          if (startDist > 0) {

            var newScale = initialScale * (dist / startDist);

            if (newScale < minScale) newScale = minScale;

            if (newScale > maxScale) newScale = maxScale;

            currentScale = newScale;

            applyImageTransform();

          }

        } else if (e.touches.length === 1 && isDragging && currentScale > 1) {

          posX = e.touches[0].clientX - startX;

          posY = e.touches[0].clientY - startY;

          applyImageTransform();

        }

      }, { passive: false });


      viewport.addEventListener('touchend', function(e) {

        if (e.touches.length < 2) startDist = 0;

        if (e.touches.length === 0) isDragging = false;

        if (currentScale <= 1) {

          currentScale = 1;

          posX = 0;

          posY = 0;

          applyImageTransform();

        }

      });

    }

    function handleRoleClick() {

      if (currentRole === 'customer') {

        var pin = prompt("សូមបញ្ចូលលេខកូដសម្ងាត់ Admin PIN (Default: 9999):");

        if (!pin) return;

        

        callBackendPost({ action: "verifyPin", pin: pin })

          .then(function(res) {

            if (res && res.valid) {

              adminPin = pin;

              localStorage.setItem('saved_admin_pin', pin);

              switchToAdmin();

            } else {

              alert("❌ លេខកូដ PIN មិនត្រឹមត្រូវ!");

            }

          .catch(function(err) { alert("Error: " + err.message); });

      } else {

        switchToCustomer();

      }

    }

    function filterAdminProducts() {
